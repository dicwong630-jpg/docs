每個 Mintlify 網站都需要一個包含核心配置設定的 `docs.json` 檔案。詳細了解以下[屬性說明](#properties)。

## 屬性說明

<ResponseField name="name" type="string" required>
你的專案名稱，用於全域標題。

範例：`mintlify`

</ResponseField>

<ResponseField name="navigation" type="Navigation[]" required>
  包含群組及其頁面的陣列
  <Expandable title="Navigation">
    <ResponseField name="group" type="string">
    群組名稱。

    範例：`Settings`

    </ResponseField>
    <ResponseField name="pages" type="string[]">
    指向 Markdown 檔案的相對路徑陣列。

    範例：`["customization", "page"]`

    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="logo" type="string or object">
  標誌圖片路徑，或包含「淺色」和「深色」模式標誌路徑的物件
  <Expandable title="Logo">
    <ResponseField name="light" type="string">
      淺色模式下的標誌路徑
    </ResponseField>
    <ResponseField name="dark" type="string">
      深色模式下的標誌路徑
    </ResponseField>
    <ResponseField name="href" type="string" default="/">
      點擊標誌後的連結目標
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="favicon" type="string">
  網站圖示圖片路徑
</ResponseField>

<ResponseField name="colors" type="Colors">
  全域主題的十六進位色碼
  <Expandable title="Colors">
    <ResponseField name="primary" type="string" required>
      主要色彩。最常用於淺色模式下的強調內容、區段標題和重點裝飾
    </ResponseField>
    <ResponseField name="light" type="string">
      深色模式的主要色彩。最常用於深色模式下的強調內容、區段標題和重點裝飾
    </ResponseField>
    <ResponseField name="dark" type="string">
      重要按鈕的主要色彩
    </ResponseField>
    <ResponseField name="background" type="object">
      淺色和深色模式下的背景色彩
      <Expandable title="Object">
        <ResponseField name="light" type="string" required>
          淺色模式下背景的十六進位色碼
        </ResponseField>
        <ResponseField name="dark" type="string" required>
          深色模式下背景的十六進位色碼
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="topbarLinks" type="TopbarLink[]">
  你想在頂部導覽列中顯示的連結名稱和網址陣列
  <Expandable title="TopbarLink">
    <ResponseField name="name" type="string">
    按鈕名稱。

    範例：`Contact us`
    </ResponseField>
    <ResponseField name="url" type="string">
    點擊按鈕後的網址。範例：`https://mintlify.com/docs`
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="topbarCtaButton" type="Call to Action">
  <Expandable title="Topbar Call to Action">
    <ResponseField name="type" type={'"link" or "github"'} default="link">
    `link` 顯示按鈕；`github` 顯示包含提供網址的儲存庫資訊，包括 GitHub 星星數量。
    </ResponseField>
    <ResponseField name="url" type="string">
    若為 `link`：按鈕連結目標。
    
    若為 `github`：用於載入 GitHub 資訊的儲存庫連結。
    </ResponseField>
    <ResponseField name="name" type="string">
    按鈕內的文字。僅在 `type` 為 `link` 時必填。
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="versions" type="string[]">
  版本名稱陣列。僅在需要於導覽列以下拉選單顯示不同版本文件時使用。
</ResponseField>

<ResponseField name="anchors" type="Anchor[]">
  錨點陣列，包含 `icon`、`color` 和 `url`。
  <Expandable title="Anchor">
    <ResponseField name="icon" type="string">
    用於展示錨點的 [Font Awesome](https://fontawesome.com/search?q=heart) 圖示。

    範例：`comments`
    </ResponseField>
    <ResponseField name="name" type="string">
    錨點標籤名稱。

    範例：`Community`
    </ResponseField>
    <ResponseField name="url" type="string">
      標記哪些頁面屬於此錨點的 URL 開頭。通常是你放置頁面的資料夾名稱。
    </ResponseField>
    <ResponseField name="color" type="string">
      錨點圖示背景的十六進位色碼。也可以傳入包含 `from` 和 `to` 屬性（各為十六進位色碼）的漸層物件。
    </ResponseField>
    <ResponseField name="version" type="string">
      若要在選擇正確文件版本前隱藏錨點，可使用此欄位。
    </ResponseField>
    <ResponseField name="isDefaultHidden" type="boolean" default="false">
      若要在直接連結到此錨點內部文件前隱藏它，請傳入 `true`。
    </ResponseField>
    <ResponseField name="iconType" default="duotone" type="string">
      可為以下其中一值："brands"、"duotone"、"light"、"sharp-solid"、"solid" 或 "thin"
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="topAnchor" type="Object">
  覆寫最頂部錨點的預設配置。
  <Expandable title="Object">
    <ResponseField name="name" default="Documentation" type="string">
      最頂部錨點的名稱
    </ResponseField>
    <ResponseField name="icon" default="book-open" type="string">
      Font Awesome 圖示。
    </ResponseField>
    <ResponseField name="iconType" default="duotone" type="string">
      可為以下其中一值："brands"、"duotone"、"light"、"sharp-solid"、"solid" 或 "thin"
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tabs" type="Tabs[]">
  導覽分頁陣列。
  <Expandable title="Tabs">
    <ResponseField name="name" type="string">
      分頁標籤名稱。
    </ResponseField>
    <ResponseField name="url" type="string">
      標記哪些頁面屬於此分頁的 URL 開頭。通常是你放置頁面的資料夾名稱。
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="api" type="API">
  API 設定配置。了解更多 [API 元件](/api-playground/demo)。
  <Expandable title="API">
    <ResponseField name="baseUrl" type="string">
      所有 API 端點的基礎 URL。若 `baseUrl` 為陣列，可啟用多個基礎 URL 選項供使用者切換。
    </ResponseField>

    <ResponseField name="auth" type="Auth">
      <Expandable title="Auth">
        <ResponseField name="method" type='"bearer" | "basic" | "key"'>
          所有 API 端點使用的驗證策略。
        </ResponseField>
        <ResponseField name="name" type="string">
        API Playground 中使用的驗證參數名稱。

        若方法為 `basic`，格式應為 `[usernameName]:[passwordName]`
        </ResponseField>
        <ResponseField name="inputPrefix" type="string">
        設計為驗證輸入欄位前綴的預設值。

        例如：若 `inputPrefix` 為 `AuthKey`，驗證欄位的預設輸入結果將繼承為 `AuthKey`。
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="playground" type="Playground">
      API Playground 的配置設定

      <Expandable title="Playground">
        <ResponseField name="mode" default="show" type='"show" | "simple" | "hide"'>
          Playground 的顯示狀態：`show` 顯示、`hide` 隱藏，或 `simple` 僅顯示端點而不提供使用者互動

          了解更多：[Playground 指南](/api-playground/demo)
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="maintainOrder" type="boolean">
      啟用此旗標可確保 OpenAPI 頁面中的鍵排序與 OpenAPI 檔案中定義的鍵排序一致。

      <Warning>此行為即將預設啟用，屆時此欄位將被棄用。</Warning>
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="openapi" type="string | string[]">
  指向你的 OpenAPI 檔案的 URL 字串或相對路徑（可為陣列）。
  
  範例：
  <CodeGroup>
    ```json 絕對路徑
    "openapi": "https://example.com/openapi.json"
    ```
    ```json 相對路徑
    "openapi": "/openapi.json"
    ```
    ```json 多個路徑
    "openapi": ["https://example.com/openapi1.json", "/openapi2.json", "/openapi3.json"]
    ```
  </CodeGroup>

</ResponseField>

<ResponseField name="footerSocials" type="FooterSocials">
  社群媒體帳號物件，鍵值對分別代表社群媒體平台和帳號網址。
  
  範例： 
  ```json
  {
    "x": "https://x.com/mintlify",
    "website": "https://mintlify.com"
  }
  ```
  <Expandable title="FooterSocials">
    <ResponseField name="[key]" type="string">
    下列其中一個值：`website`、`facebook`、`x`、`discord`、`slack`、`github`、`linkedin`、`instagram`、`hacker-news`
    
    範例：`x`
    </ResponseField>
    <ResponseField name="property" type="string">
    社群平台的 URL。
    
    範例：`https://x.com/mintlify`
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="feedback" type="Feedback">
  啟用回饋按鈕的配置設定

  <Expandable title="Feedback">
    <ResponseField name="suggestEdit" type="boolean" default="false">
    啟用允許使用者透過 Pull Request 建議編輯的按鈕
    </ResponseField>
    <ResponseField name="raiseIssue" type="boolean" default="false">
    啟用允許使用者提出文件問題的按鈕
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="modeToggle" type="ModeToggle">
  自訂深色模式切換按鈕。
  <Expandable title="ModeToggle">
    <ResponseField name="default" type={'"light" or "dark"'}>
      設定新使用者始終顯示淺色或深色模式。若未設定，預設跟隨使用者作業系統的模式。
    </ResponseField>
    <ResponseField name="isHidden" type="boolean" default="false">
      設為 true 可隱藏深色/淺色模式切換按鈕。可搭配 `default` 強制只使用淺色或深色模式。例如：
      
      <CodeGroup>
      ```json 僅深色模式
      "modeToggle": {
        "default": "dark",
        "isHidden": true
      }
      ```

      ```json 僅淺色模式
      "modeToggle": {
        "default": "light",
        "isHidden": true
      }
      ```
      </CodeGroup>

    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="backgroundImage" type="string">
  在每個頁面背後顯示的背景圖片。範例可參考 [Infisical](https://infisical.com/docs) 和 [FRPC](https://frpc.io)。
</ResponseField>
